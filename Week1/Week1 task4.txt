class Solution {
    public int[] getSumAbsoluteDifferences(int[] nums) {
        int n = nums.length;
        int[] result = new int[n];

        long totalSum = 0;

        // Find total sum
        for (int num : nums) {
            totalSum += num;
        }

        long prefixSum = 0;

        for (int i = 0; i < n; i++) {
            long current = nums[i];

            // Sum of differences with elements on the left
            long left = current * i - prefixSum;

            // Sum of differences with elements on the right
            long right = (totalSum - prefixSum - current)
                       - current * (n - i - 1);

            result[i] = (int) (left + right);

            // Add current element to prefix sum
            prefixSum += current;
        }

        return result;
    }
}
